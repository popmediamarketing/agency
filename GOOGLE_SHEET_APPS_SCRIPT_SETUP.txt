function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetName = 'Engagement Ring Leads';
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);

  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'Timestamp',
      'Name',
      'Phone',
      'Story',
      'Proposal Timeline',
      'Budget',
      'Lifestyle',
      'Priority',
      'Suitable Setting',
      'Setting Series',
      'Setting Score',
      'Suitable Diamond',
      'Suitable Material',
      'Suitable Width',
      'Suitable Warranty',
      'Suitable After Sales',
      'Suitable Budget Allocation',
      'Timeline Advice',
      'Source'
    ]);
  }

  var data = JSON.parse(e.postData.contents || '{}');
  var rec = data.recommendations || {};

  sheet.appendRow([
    new Date(),
    data.name || '',
    data.phone || '',
    data.story || '',
    data.timeline || '',
    data.budget || '',
    data.lifestyle || '',
    data.priority || '',
    rec['Suitable Setting'] || '',
    rec['Setting Series'] || '',
    rec['Setting Score'] || '',
    rec['Suitable Diamond'] || '',
    rec['Suitable Material'] || '',
    rec['Suitable Width'] || '',
    rec['Suitable Warranty'] || '',
    rec['Suitable After Sales'] || '',
    rec['Suitable Budget Allocation'] || '',
    rec['Timeline Advice'] || '',
    data.source || 'Engagement Ring AI Advisor'
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
